cmake_minimum_required(VERSION 3.22)
project(sing_native_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
option(SING_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(SING_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(DSP ${ROOT}/app/src/main/cpp)
set(SPEEX ${ROOT}/extern/speexdsp)
add_library(speex_test STATIC ${SPEEX}/libspeexdsp/resample.c)
set_target_properties(speex_test PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(speex_test PUBLIC ${SPEEX}/include/speex)
target_compile_definitions(speex_test PUBLIC OUTSIDE_SPEEX RANDOM_PREFIX=sing_speex)
target_compile_definitions(speex_test PRIVATE FLOATING_POINT)
add_library(kiss_test STATIC ${ROOT}/extern/kissfft/kiss_fft.c ${ROOT}/extern/kissfft/kiss_fftr.c)
set_target_properties(kiss_test PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(kiss_test PUBLIC ${ROOT}/extern/kissfft)
target_compile_definitions(kiss_test PUBLIC kiss_fft_scalar=float)
add_library(dsp_test STATIC ${DSP}/stft_engine.cpp ${DSP}/vr_dsp_engine.cpp ${DSP}/speex_resampler.cpp)
set_target_properties(dsp_test PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(dsp_test PUBLIC ${DSP})
target_link_libraries(dsp_test PUBLIC kiss_test speex_test Threads::Threads)
enable_testing()
add_executable(center_cut_test center_cut_test.cpp)
target_link_libraries(center_cut_test PRIVATE dsp_test)
add_test(NAME center_cut COMMAND center_cut_test)
add_executable(resampler_test resampler_test.cpp)
target_link_libraries(resampler_test PRIVATE dsp_test)
add_test(NAME resampler COMMAND resampler_test)

# Optional desktop JNI build for JVM streaming/concurrency regression tests.
find_package(JNI)
if(JNI_FOUND)
    add_library(sing_dsp SHARED ${DSP}/jni_bridge.cpp)
    target_include_directories(sing_dsp PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(sing_dsp PRIVATE dsp_test)
endif()
